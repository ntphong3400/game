const _jade = Color(0xFF16372F);
const _deepJade = Color(0xFF102922);
const _gilt = Color(0xFFD9B873);
const _ivory = Color(0xFFFFF1CE);
const _son = Color(0xFF922F28);

/// Giang Sơn overview. The internal citadel route stays compatible with saves.
/// All actions are delegated to the existing game shell.
class CitadelHub extends StatelessWidget {
  const CitadelHub({
    super.key,
    required this.state,
    required this.busy,
    required this.onNavigate,
    required this.onMenu,
    required this.onEndTurn,
    required this.onOvertime,
  });

  final GameState state;
  final bool busy;
  final ValueChanged<String> onNavigate;
  final VoidCallback onMenu, onEndTurn, onOvertime;

  @override
  Widget build(BuildContext context) => DefaultTextStyle(
    style: const TextStyle(
      fontFamily: 'BeVietnam',
      color: _ivory,
      fontSize: 13,
    ),
    child: SafeArea(
      child: ColoredBox(
        color: _deepJade,
        child: LayoutBuilder(
          builder:
              (context, bounds) => Column(
                children: [
                  _header(bounds.maxWidth >= 720, bounds.maxHeight < 520),
                  if (busy)
                    const LinearProgressIndicator(
                      minHeight: 2,
                      color: _gilt,
                      backgroundColor: _deepJade,
                    ),
                  Expanded(
                    child: Stack(
                      children: [
                        Positioned.fill(
                          child: _NationalMap(
                            state: state,
                            onPalace: () => _openPalace(context),
                            onNavigate: onNavigate,
                          ),
                        ),
                        if (bounds.maxWidth >= 720)
                          Positioned(
                            top: 16,
                            right: 16,
                            child: IgnorePointer(
                              child: Container(
                                padding: const EdgeInsets.symmetric(
                                  horizontal: 18,
                                  vertical: 10,
                                ),
                                decoration: BoxDecoration(
                                  color: _deepJade.withValues(alpha: .9),
                                  border: Border.all(color: _gilt),
                                  borderRadius: BorderRadius.circular(4),
                                ),
                                child: const Column(
                                  mainAxisSize: MainAxisSize.min,
                                  children: [
                                    Text(
                                      'GIANG SƠN',
                                      style: TextStyle(
                                        fontFamily: 'Lora',
                                        fontSize: 20,
                                        letterSpacing: 2,
                                        color: _ivory,
                                      ),
                                    ),
                                    Text(
                                      'Lãnh thổ của bệ hạ',
                                      style: TextStyle(
                                        fontSize: 10,
                                        color: _gilt,
                                      ),
                                    ),
                                  ],
                                ),
                              ),
                            ),
                          ),
                        if (bounds.maxWidth < 720)
                          const Positioned(
                            left: 12,
                            bottom: 20,
                            child: IgnorePointer(
                              child: Text(
                                'Kéo ngang để xem bản đồ',
                                style: TextStyle(
                                  fontSize: 10,
                                  color: _ivory,
                                  shadows: [
                                    Shadow(color: _deepJade, blurRadius: 8),
                                  ],
                                ),
                              ),
                            ),
                          ),
                        Positioned(
                          right: 20,
                          bottom: 18,
                          width: bounds.maxWidth < 600 ? 168 : 205,
                          child: Tooltip(
                            message: '${state.points} hành động còn lại',
                            child: _PalaceButton(
                              key: const ValueKey('citadel-end-turn'),
                              onPressed: busy ? null : onEndTurn,
                              label: state.seasonal ? 'Khóa mùa' : 'Sang tuần',
                              icon: Icons.arrow_forward,
                              primary: true,
                              compact: bounds.maxHeight < 520,
                            ),
                          ),
                        ),
                      ],
                    ),
                  ),
                ],
              ),
        ),
      ),
    ),
  );

  Future<void> _openPalace(BuildContext context) => showModalBottomSheet<void>(
    context: context,
    backgroundColor: _deepJade,
    showDragHandle: true,
    isScrollControlled: true,
    builder:
        (sheetContext) => SafeArea(
          child: SingleChildScrollView(
            padding: const EdgeInsets.fromLTRB(20, 0, 20, 24),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Row(
                  children: [
                    const Expanded(
                      child: Text(
                        'Hoàng cung',
                        style: TextStyle(
                          fontFamily: 'Lora',
                          fontSize: 24,
                          color: _ivory,
                        ),
                      ),
                    ),
                    IconButton(
                      tooltip: 'Đóng Hoàng cung',
                      onPressed: () => Navigator.pop(sheetContext),
                      icon: const Icon(Icons.close, color: _gilt),
                    ),
                  ],
                ),
                Text(
                  '${state.petitions.length} tấu chương chờ xử lý · ${state.points} hành động còn lại',
                  style: const TextStyle(fontSize: 12, color: _gilt),
                ),
                const SizedBox(height: 16),
                Wrap(
                  spacing: 12,
                  runSpacing: 12,
                  children: [
                    for (final item in const [
                      ('court', 'Triều chính', Icons.account_balance_outlined),
                      ('treasury', 'Ngân khố', Icons.savings_outlined),
                      ('people', 'Triều thần', Icons.groups_outlined),
                      ('harem', 'Hậu cung', Icons.local_florist_outlined),
                      ('health', 'Long thể', Icons.spa_outlined),
                    ])
                      SizedBox(
                        width: math.min(
                          220,
                          (MediaQuery.sizeOf(sheetContext).width - 52) / 2,
                        ),
                        child: _PalaceButton(
                          key: ValueKey('citadel-location-${item.$1}'),
                          label: item.$2,
                          icon: item.$3,
                          art: item.$1,
                          count:
                              item.$1 == 'court' ? state.petitions.length : 0,
                          onPressed: () {
                            Navigator.pop(sheetContext);
                            onNavigate(item.$1);
                          },
                        ),
                      ),
                  ],
                ),
                if (state.points == 0 && state.overtime == 0)
                  TextButton(
                    onPressed:
                        busy
                            ? null
                            : () {
                              Navigator.pop(sheetContext);
                              onOvertime();
                            },
                    child: const Text(
                      'Chong đèn làm thêm',
                      style: TextStyle(color: _gilt),
                    ),
                  ),
              ],
            ),
          ),
        ),
  );

  Widget _header(bool wide, bool short) {
    final identity = Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        IconButton(
          tooltip: 'Mở danh mục',
          onPressed: onMenu,
          icon: const Icon(Icons.menu, color: _gilt, size: 21),
        ),
        Container(
          width: short || !wide ? 40 : 50,
          height: short || !wide ? 40 : 50,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            border: Border.all(color: _gilt, width: 2),
            gradient: const LinearGradient(colors: [_jade, _deepJade]),
          ),
          child: const ImperialIcon('authority', size: 44),
        ),
        const SizedBox(width: 12),
        Flexible(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(
                'VẠN TUẾ',
                style: TextStyle(
                  fontFamily: 'Lora',
                  fontSize: short ? 16 : 20,
                  letterSpacing: short ? 1 : 3,
                  color: _ivory,
                ),
              ),
              Text(
                wide ? state.playerName : 'Giang Sơn · ${state.playerName}',
                maxLines: 1,
                overflow: TextOverflow.ellipsis,
                style: const TextStyle(color: _gilt, fontSize: 12),
              ),
              if (!short)
                Text(
                  'Năm trị vì ${state.year} · ${state.age} tuổi',
                  style: const TextStyle(fontSize: 10, color: _ivory),
                ),
            ],
          ),
        ),
      ],
    );
    final calendar = Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.end,
      children: [
        Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            Icon(
              state.season == 'Đông'
                  ? Icons.ac_unit
                  : Icons.filter_vintage_outlined,
              color: _gilt,
              size: 18,
            ),
            const SizedBox(width: 6),
            Flexible(
              child: Text(
                'Mùa ${state.season}',
                maxLines: 1,
                overflow: TextOverflow.ellipsis,
                style: const TextStyle(fontFamily: 'Lora', fontSize: 16),
              ),
            ),
          ],
        ),
        const SizedBox(height: 4),
        Text(
          state.seasonal
              ? 'Thiên Hòa · năm ${state.year}'
              : 'Năm ${state.year} · tuần ${(state.completedWeeks % 52) + 1}',
          style: const TextStyle(fontSize: 10, color: _gilt),
        ),
        Text(
          '${state.points} hành động còn lại',
          key: const ValueKey('citadel-actions'),
          style: const TextStyle(fontSize: 10, color: _ivory),
        ),
      ],
    );
    return Container(
      padding: EdgeInsets.fromLTRB(4, short ? 6 : 12, 18, short ? 6 : 12),
      decoration: const BoxDecoration(
        gradient: LinearGradient(colors: [_deepJade, _jade, _deepJade]),
        image: DecorationImage(
          image: AssetImage('assets/images/imperial/jade-texture.webp'),
          fit: BoxFit.cover,
          opacity: .25,
        ),
        border: Border(bottom: BorderSide(color: _gilt)),
      ),
      child: Column(
        children: [
          Row(
            children: [
              SizedBox(
                width: wide ? (short ? 190 : 220) : null,
                child: wide ? identity : null,
              ),
              if (!wide) Expanded(child: identity),
              if (wide) Expanded(child: _stats(short: short)),
              const SizedBox(width: 12),
              SizedBox(
                width: wide ? (short ? 140 : 160) : 125,
                child: calendar,
              ),
            ],
          ),
          if (!wide) ...[const SizedBox(height: 12), _stats(short: false)],
        ],
      ),
    );
  }

  Widget _stats({required bool short}) {
    final values = <(String, int, IconData, String)>[
      ('Ngân khố', state.treasury, Icons.savings_outlined, 'treasury'),
      ('Lòng dân', state.people, Icons.groups_outlined, 'court'),
      ('Uy quyền', state.authority, Icons.workspace_premium_outlined, 'court'),
      (
        'Quân lực',
        militaryStrength(state),
        Icons.shield_outlined,
        state.population == null ? 'kingdom' : 'military',
      ),
      ('Sức khỏe', state.health, Icons.favorite_border, 'health'),
      ('Ổn định triều đình', state.stability, Icons.balance_outlined, 'court'),
    ];
    return LayoutBuilder(
      builder:
          (context, bounds) => Wrap(
            alignment: WrapAlignment.center,
            runSpacing: 6,
            children: [
              for (final stat in values)
                SizedBox(
                  width: bounds.maxWidth / (bounds.maxWidth < 440 ? 3 : 6),
                  child: Tooltip(
                    message:
                        '${stat.$1}: ${stat.$2}${stat.$4 == 'treasury'
                            ? ' ${moneyUnit(state)}'
                            : stat.$4 == 'military'
                            ? ' điểm'
                            : '/100'} · Nhấn để xem chi tiết',
                    child: InkWell(
                      key: ValueKey('citadel-stat-${stat.$1}'),
                      onTap: () => onNavigate(stat.$4),
                      borderRadius: BorderRadius.circular(6),
                      child: Padding(
                        padding: const EdgeInsets.symmetric(
                          vertical: 3,
                          horizontal: 4,
                        ),
                        child: Column(
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            Row(
                              mainAxisAlignment: MainAxisAlignment.center,
                              children: [
                                ImperialIcon(switch (stat.$1) {
                                  'Ngân khố' => 'treasury',
                                  'Lòng dân' => 'citizens',
                                  'Uy quyền' => 'authority',
                                  'Quân lực' => 'army',
                                  'Sức khỏe' => 'health',
                                  _ => 'stability',
                                }, size: short ? 24 : 30),
                                const SizedBox(width: 6),
                                Flexible(
                                  child: FittedBox(
                                    fit: BoxFit.scaleDown,
                                    child: Text(
                                      stat.$4 == 'treasury' &&
                                              state.goldStandard
                                          ? moneyNumber(stat.$2)
                                          : '${stat.$2}',
                                      maxLines: 1,
                                      style: TextStyle(
                                        fontFamily: 'Lora',
                                        fontSize: short ? 16 : 19,
                                        color:
                                            stat.$4 != 'treasury' &&
                                                    stat.$2 <= 20
                                                ? const Color(0xFFFFAA90)
                                                : _ivory,
                                      ),
                                    ),
                                  ),
                                ),
                              ],
                            ),
                            const SizedBox(height: 3),
                            Text(
                              stat.$1,
                              textAlign: TextAlign.center,
                              style: const TextStyle(
                                fontSize: 10,
                                color: _ivory,
                              ),
                            ),
                          ],
                        ),
                      ),
                    ),
                  ),
                ),
            ],
          ),
    );
  }
}

/// The complete map can be dragged horizontally on narrow or tall screens.
/// Artwork and labels share one canvas, so landmarks never drift when panning.
class _NationalMap extends StatefulWidget {
  const _NationalMap({
    required this.state,
    required this.onPalace,
    required this.onNavigate,
  });
  final GameState state;
  final VoidCallback onPalace;
  final ValueChanged<String> onNavigate;
  @override
  State<_NationalMap> createState() => _NationalMapState();
}

class _NationalMapState extends State<_NationalMap> {
  ScrollController? _scroll;
  @override
  void dispose() {
    _scroll?.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => LayoutBuilder(
    builder: (context, bounds) {
      // This matches the aspect ratio of the shipped background plate.
      final width =
          bounds.maxWidth >= 720
              ? bounds.maxWidth
              : math.max(bounds.maxWidth, bounds.maxHeight * (1922 / 818));
      _scroll ??= ScrollController(
        initialScrollOffset: (width - bounds.maxWidth) * .48,
      );
      final military = widget.state.population == null ? 'kingdom' : 'military';
      final neighbors = widget.state.warfare?.neighbors;
      String country(String id, String fallback) =>
          neighbors?.where((n) => n.id == id).firstOrNull?.name ?? fallback;
      final landmarks = <(String, String, Offset, VoidCallback)>[
        ('palace', 'Hoàng cung', const Offset(.49, .36), widget.onPalace),
        (
          'capital',
          'Kinh thành',
          const Offset(.64, .50),
          () => widget.onNavigate(
            widget.state.population == null ? 'kingdom' : 'population',
          ),
        ),
        (
          'farms',
          'Ruộng · Đê điều',
          const Offset(.76, .67),
          () => widget.onNavigate('treasury'),
        ),
        (
          'north-pass',
          'Bắc ải',
          const Offset(.525, .15),
          () => widget.onNavigate(military),
        ),
        (
          'west-pass',
          'Tây ải',
          const Offset(.27, .46),
          () => widget.onNavigate(military),
        ),
        (
          'south-pass',
          'Nam ải',
          const Offset(.595, .81),
          () => widget.onNavigate(military),
        ),
        (
          'north',
          country('north', 'Hàn Kỳ'),
          const Offset(.26, .19),
          () => widget.onNavigate(military),
        ),
        (
          'west',
          country('west', 'Xích Lĩnh'),
          const Offset(.10, .57),
          () => widget.onNavigate(military),
        ),
        (
          'south',
          country('south', 'Bích Trạch'),
          const Offset(.29, .90),
          () => widget.onNavigate(military),
        ),
      ];
      return ScrollConfiguration(
        behavior: ScrollConfiguration.of(
          context,
        ).copyWith(dragDevices: PointerDeviceKind.values.toSet()),
        child: Scrollbar(
          controller: _scroll,
          child: SingleChildScrollView(
            key: const ValueKey('giang-son-map-scroll'),
            controller: _scroll,
            scrollDirection: Axis.horizontal,
            child: SizedBox(
              width: width,
              height: bounds.maxHeight,
              child: Stack(
                key: const ValueKey('giang-son-map'),
                fit: StackFit.expand,
                children: [
                  Image.asset(
                    'assets/images/giang-son-map.webp',
                    fit: BoxFit.fill,
                    excludeFromSemantics: true,
                  ),
                  for (final landmark in landmarks)
                    Positioned(
                      left: (width * landmark.$3.dx -
                              (bounds.maxWidth >= 720 && bounds.maxHeight < 360
                                  ? 58
                                  : 75))
                          .clamp(4.0, width - 154),
                      top: (bounds.maxHeight * landmark.$3.dy - 42).clamp(
                        4.0,
                        math.max(4.0, bounds.maxHeight - 52),
                      ),
                      width:
                          bounds.maxWidth >= 720 && bounds.maxHeight < 360
                              ? 116
                              : 150,
                      child: _MapLabel(
                        key: ValueKey('giang-son-${landmark.$1}'),
                        label: landmark.$2,
                        onPressed: landmark.$4,
                        compact: bounds.maxHeight < 360,
                        count:
                            landmark.$1 == 'palace'
                                ? widget.state.petitions.length
                                : 0,
                      ),
                    ),
                ],
              ),
            ),
          ),
        ),
      );
    },
  );
}

class _MapLabel extends StatelessWidget {
  const _MapLabel({
    super.key,
    required this.label,
    required this.onPressed,
    this.count = 0,
    this.compact = false,
  });
  final String label;
  final VoidCallback onPressed;
  final int count;
  final bool compact;
  @override
  Widget build(BuildContext context) => Material(
    color: _deepJade.withValues(alpha: .95),
    elevation: 5,
    borderRadius: BorderRadius.circular(4),
    child: InkWell(
      onTap: onPressed,
      borderRadius: BorderRadius.circular(4),
      child: Container(
        constraints: const BoxConstraints(minHeight: 44),
        padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 6),
        decoration: BoxDecoration(
          border: Border.all(color: _gilt),
          borderRadius: BorderRadius.circular(4),
        ),
        child: Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Flexible(
              child: Text(
                label,
                textAlign: TextAlign.center,
                style: TextStyle(
                  fontFamily: 'Lora',
                  fontSize: compact ? 12 : 15,
                  color: _ivory,
                ),
              ),
            ),
            if (count > 0) ...[
              const SizedBox(width: 6),
              Badge(
                label: Text('$count'),
                backgroundColor: _son,
                textColor: _ivory,
              ),
            ],
          ],
        ),
      ),
    ),
  );
}

class _PalaceButton extends StatefulWidget {
  const _PalaceButton({
    super.key,
    required this.onPressed,
    required this.label,
    required this.icon,
    this.art,
    this.count = 0,
    this.primary = false,
    this.compact = false,
  });
  final VoidCallback? onPressed;
  final String label;
  final IconData icon;
  final String? art;
  final int count;
  final bool primary, compact;

  @override
  State<_PalaceButton> createState() => _PalaceButtonState();
}

class _PalaceButtonState extends State<_PalaceButton> {
  bool hovered = false, focused = false;

  @override
  Widget build(BuildContext context) => MouseRegion(
    onEnter: (_) => setState(() => hovered = true),
    onExit: (_) => setState(() => hovered = false),
    child: AnimatedContainer(
      duration: const Duration(milliseconds: 140),
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(widget.primary ? 10 : 5),
        border: focused ? Border.all(color: _ivory, width: 2) : null,
        boxShadow: [
          BoxShadow(
            color:
                hovered || focused
                    ? _gilt.withValues(alpha: .35)
                    : const Color(0x66000000),
            blurRadius: hovered || focused ? 16 : 6,
            offset: const Offset(0, 3),
          ),
        ],
      ),
      child: Material(
        color: Colors.transparent,
        borderRadius: BorderRadius.circular(widget.primary ? 9 : 4),
        child: InkWell(
          onTap: widget.onPressed,
          onFocusChange: (value) => setState(() => focused = value),
          borderRadius: BorderRadius.circular(widget.primary ? 9 : 4),
          child: Opacity(
            opacity: widget.onPressed == null ? .45 : 1,
            child: ImperialSkin(
              asset: widget.primary ? 'crimson' : 'jade',
              edge: 20,
              child: ConstrainedBox(
                constraints: const BoxConstraints(minHeight: 54),
                child: Padding(
                  padding: EdgeInsets.symmetric(
                    horizontal: 12,
                    vertical: widget.compact ? 8 : 12,
                  ),
                  child: Row(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      if (widget.art != null)
                        ImperialIcon(widget.art!, size: 32)
                      else
                        Icon(widget.icon, color: _gilt, size: 22),
                      const SizedBox(width: 8),
                      Flexible(
                        child: Text(
                          widget.label,
                          textAlign: TextAlign.center,
                          style: TextStyle(
                            fontFamily: 'Lora',
                            color: _ivory,
                            fontSize: widget.primary ? 19 : 15,
                          ),
                        ),
                      ),
                      if (widget.count > 0) ...[
                        const SizedBox(width: 8),
                        Container(
                          padding: const EdgeInsets.symmetric(
                            horizontal: 6,
                            vertical: 2,
                          ),
                          decoration: BoxDecoration(
                            color: _son,
                            borderRadius: BorderRadius.circular(3),
                            border: Border.all(color: _gilt),
                          ),
                          child: Text(
                            '${widget.count}',
                            style: const TextStyle(color: _ivory, fontSize: 12),
                          ),
                        ),
                      ],
                    ],
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    ),
  );
}
