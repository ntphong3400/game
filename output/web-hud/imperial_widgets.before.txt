/// One shared HUD for all five interior screens. Reading it never dispatches commands.
class ImperialStatusBar extends StatelessWidget {
  const ImperialStatusBar({
    super.key,
    required this.state,
    required this.title,
    required this.onMenu,
    required this.onHome,
    required this.onNavigate,
    required this.desktop,
  });
  final GameState state;
  final String title;
  final VoidCallback onMenu, onHome;
  final ValueChanged<String> onNavigate;
  final bool desktop;

  @override
  Widget build(BuildContext context) => Container(
    padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 9),
    decoration: const BoxDecoration(
      color: imperialDeep,
      border: Border(bottom: BorderSide(color: imperialGold)),
    ),
    child: LayoutBuilder(
      builder: (context, bounds) {
        final roomy = bounds.maxWidth >= 740;
        final identity = Row(
          children: [
            IconButton(
              tooltip: desktop ? 'Về Giang Sơn' : 'Mở danh mục',
              onPressed: desktop ? onHome : onMenu,
              icon: Icon(
                desktop ? Icons.castle_outlined : Icons.menu,
                color: imperialGold,
                size: 22,
              ),
            ),
            if (!desktop)
              IconButton(
                tooltip: 'Về Giang Sơn',
                onPressed: onHome,
                icon: const Icon(
                  Icons.castle_outlined,
                  color: imperialGold,
                  size: 22,
                ),
              ),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(
                    title,
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: const TextStyle(
                      fontFamily: 'Lora',
                      fontSize: 18,
                      color: imperialIvory,
                    ),
                  ),
                  Text(
                    '${state.playerName} · ${state.seasonal ? 'Mùa ${state.season}' : 'Tuần ${state.week}'} · năm ${state.year}',
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: const TextStyle(fontSize: 10, color: imperialGold),
                  ),
                ],
              ),
            ),
          ],
        );
        final stats = <(String, int, IconData, String)>[
          ('Ngân khố', state.treasury, Icons.savings_outlined, 'treasury'),
          ('Lòng dân', state.people, Icons.groups_outlined, 'court'),
          (
            'Uy quyền',
            state.authority,
            Icons.workspace_premium_outlined,
            'court',
          ),
          (
            'Quân lực',
            militaryStrength(state),
            Icons.shield_outlined,
            state.population == null ? 'kingdom' : 'military',
          ),
          ('Sức khỏe', state.health, Icons.favorite_border, 'health'),
          (
            'Ổn định triều đình',
            state.stability,
            Icons.balance_outlined,
            'court',
          ),
        ];
        final values = LayoutBuilder(
          builder:
              (context, box) => Row(
                children: [
                  for (final stat in stats)
                    Expanded(
                      child: Tooltip(
                        message:
                            '${stat.$1}: ${stat.$2}${stat.$4 == 'treasury'
                                ? ' ${moneyUnit(state)}'
                                : stat.$4 == 'military'
                                ? ' điểm'
                                : '/100'}',
                        child: InkWell(
                          onTap: () => onNavigate(stat.$4),
                          borderRadius: BorderRadius.circular(5),
                          child: Padding(
                            padding: const EdgeInsets.symmetric(
                              vertical: 5,
                              horizontal: 2,
                            ),
                            child: Column(
                              mainAxisSize: MainAxisSize.min,
                              children: [
                                Icon(
                                  stat.$3,
                                  color: imperialGold,
                                  size: roomy ? 18 : 16,
                                ),
                                Text(
                                  stat.$4 == 'treasury' && state.goldStandard
                                      ? moneyNumber(stat.$2)
                                      : '${stat.$2}',
                                  style: TextStyle(
                                    fontFamily: 'Lora',
                                    fontSize: roomy ? 17 : 15,
                                    color:
                                        stat.$4 != 'treasury' && stat.$2 <= 20
                                            ? const Color(0xFFFFAA90)
                                            : imperialIvory,
                                  ),
                                ),
                                Text(
                                  stat.$1,
                                  maxLines: roomy ? 1 : 2,
                                  textAlign: TextAlign.center,
                                  style: const TextStyle(
                                    fontSize: 9,
                                    color: imperialIvory,
                                  ),
                                ),
                              ],
                            ),
                          ),
                        ),
                      ),
                    ),
                ],
              ),
        );
        return roomy
            ? Row(
              children: [
                SizedBox(width: 240, child: identity),
                Expanded(child: values),
              ],
            )
            : Column(children: [identity, const SizedBox(height: 4), values]);
      },
    ),
  );
}

/// Each identity has one three-cell atlas; age selects the cell, never the face.
class CharacterPortrait extends StatelessWidget {
  const CharacterPortrait({
    super.key,
    this.person,
    this.emperor = false,
    this.emperorAge = 18,
    this.width = 88,
    this.height = 108,
  });
  final Courtier? person;
  final bool emperor;
  final int emperorAge;
  final double width, height;

  String get portraitId =>
      emperor ? emperorPortraitId : person!.resolvedPortraitId;
  PortraitAge get portraitAge =>
      portraitAgeFor(emperor ? emperorAge : person!.age);

  @override
  Widget build(BuildContext context) {
    final stage = portraitAge;
    return Semantics(
      image: true,
      label:
          emperor
              ? 'Chân dung hoàng đế · ${stage.label}'
              : 'Chân dung ${person!.name} · ${stage.label}',
      child: Container(
        width: width,
        height: height,
        clipBehavior: Clip.antiAlias,
        decoration: BoxDecoration(
          color: const Color(0xFFE7D8BA),
          borderRadius: BorderRadius.circular(6),
          border: Border.all(color: const Color(0xFFB89559)),
        ),
        child: LayoutBuilder(
          builder: (context, bounds) {
            final side = math.max(bounds.maxWidth, bounds.maxHeight);
            return Stack(
              clipBehavior: Clip.hardEdge,
              children: [
                Positioned(
                  left: bounds.maxWidth / 2 - (stage.index + .5) * side,
                  top: (bounds.maxHeight - side) / 2,
                  width: side * 3,
                  height: side,
                  child: Image.asset(
                    portraitAsset(portraitId),
                    key: ValueKey('$portraitId-${stage.name}'),
                    fit: BoxFit.fill,
                    cacheWidth: 1152,
                    excludeFromSemantics: true,
                    gaplessPlayback: true,
                    errorBuilder:
                        (_, _, _) => Row(
                          children: List.generate(
                            3,
                            (_) => const Expanded(
                              child: ColoredBox(
                                color: Color(0xFFE7D8BA),
                                child: Center(
                                  child: Icon(
                                    Icons.person,
                                    color: imperialJade,
                                  ),
                                ),
                              ),
                            ),
                          ),
                        ),
                  ),
                ),
              ],
            );
          },
        ),
      ),
    );
  }
}

class ImperialSplit extends StatelessWidget {
  const ImperialSplit({
    super.key,
    required this.main,
    required this.aside,
    this.breakpoint = 900,
    this.asideWidth = 300,
  });
  final Widget main, aside;
  final double breakpoint, asideWidth;

  @override
  Widget build(BuildContext context) => LayoutBuilder(
    builder:
        (context, bounds) =>
            bounds.maxWidth < breakpoint
                ? Column(
                  crossAxisAlignment: CrossAxisAlignment.stretch,
                  children: [main, const SizedBox(height: 22), aside],
                )
                : Row(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Expanded(child: main),
                    const SizedBox(width: 22),
                    SizedBox(width: asideWidth, child: aside),
                  ],
                ),
  );
}
