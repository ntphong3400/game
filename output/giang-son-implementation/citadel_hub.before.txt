import '../../logic/currency_rules.dart';
import '../../logic/population_rules.dart';
import 'dart:math' as math;
import 'package:flutter/material.dart';
import '../../models/game_state.dart';
import 'imperial_assets.dart';

const _jade = Color(0xFF16372F);
const _deepJade = Color(0xFF102922);
const _gilt = Color(0xFFD9B873);
const _ivory = Color(0xFFFFF1CE);
const _son = Color(0xFF922F28);

/// Navigation only. All actions are delegated to the existing game shell.
class CitadelHub extends StatelessWidget {
  const CitadelHub({
    super.key,
    required this.state,
    required this.busy,
    required this.onNavigate,
    required this.onMenu,
    required this.onEndTurn,
    required this.onOvertime,
  });

  final GameState state;
  final bool busy;
  final ValueChanged<String> onNavigate;
  final VoidCallback onMenu, onEndTurn, onOvertime;

  static const _locations = <(String, String, IconData, Offset)>[
    ('court', 'Triều chính', Icons.account_balance_outlined, Offset(.51, .29)),
    ('treasury', 'Ngân khố', Icons.savings_outlined, Offset(.16, .40)),
    ('people', 'Triều thần', Icons.groups_outlined, Offset(.16, .66)),
    ('harem', 'Hậu cung', Icons.local_florist_outlined, Offset(.83, .38)),
    ('health', 'Long thể', Icons.spa_outlined, Offset(.87, .58)),
  ];

  @override
  Widget build(BuildContext context) => DefaultTextStyle(
    style: const TextStyle(
      fontFamily: 'BeVietnam',
      color: _ivory,
      fontSize: 13,
    ),
    child: SafeArea(
      child: LayoutBuilder(
        builder: (context, bounds) {
          final wide = bounds.maxWidth >= 720;
          final short = bounds.maxHeight < 520;
          final content = Column(
            children: [
              _header(wide, short),
              if (busy)
                const LinearProgressIndicator(
                  minHeight: 2,
                  color: _gilt,
                  backgroundColor: _deepJade,
                ),
              if (wide) ...[
                Expanded(child: _scene(showLocations: true)),
                _footer(wide: true, short: short),
              ] else ...[
                AspectRatio(
                  aspectRatio: 16 / 10,
                  child: _scene(showLocations: false),
                ),
                Padding(
                  padding: const EdgeInsets.fromLTRB(16, 18, 16, 8),
                  child: Wrap(
                    spacing: 12,
                    runSpacing: 6,
                    children: [
                      const SizedBox(
                        child: Text(
                          'CHỐN HOÀNG THÀNH',
                          style: TextStyle(
                            letterSpacing: 2,
                            fontSize: 11,
                            color: _gilt,
                          ),
                        ),
                      ),
                      const Icon(
                        Icons.screen_rotation_outlined,
                        color: _gilt,
                        size: 16,
                      ),
                      const SizedBox(width: 6),
                      const Text(
                        'Xoay ngang để ngắm cung điện',
                        style: TextStyle(fontSize: 10, color: _ivory),
                      ),
                    ],
                  ),
                ),
                Padding(
                  padding: const EdgeInsets.all(12),
                  child: Wrap(
                    spacing: 8,
                    runSpacing: 8,
                    children: [
                      for (final location in _locations)
                        SizedBox(
                          width: (bounds.maxWidth - 32) / 2,
                          child: _locationButton(location, compact: true),
                        ),
                    ],
                  ),
                ),
                _footer(wide: false, short: false),
              ],
            ],
          );
          return ColoredBox(
            color: _deepJade,
            child: wide ? content : SingleChildScrollView(child: content),
          );
        },
      ),
    ),
  );

  Widget _header(bool wide, bool short) {
    final identity = Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        IconButton(
          tooltip: 'Mở danh mục',
          onPressed: onMenu,
          icon: const Icon(Icons.menu, color: _gilt, size: 21),
        ),
        Container(
          width: short ? 40 : 50,
          height: short ? 40 : 50,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            border: Border.all(color: _gilt, width: 2),
            gradient: const LinearGradient(colors: [_jade, _deepJade]),
          ),
          child: const ImperialIcon('authority', size: 44),
        ),
        const SizedBox(width: 12),
        Flexible(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text(
                'VẠN TUẾ',
                style: TextStyle(
                  fontFamily: 'Lora',
                  fontSize: 20,
                  letterSpacing: 3,
                  color: _ivory,
                ),
              ),
              Text(
                state.playerName,
                maxLines: 1,
                overflow: TextOverflow.ellipsis,
                style: const TextStyle(color: _gilt, fontSize: 12),
              ),
              if (!short)
                Text(
                  'Năm trị vì ${state.year} · ${state.age} tuổi',
                  style: const TextStyle(fontSize: 10, color: _ivory),
                ),
            ],
          ),
        ),
      ],
    );
    final calendar = Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.end,
      children: [
        Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            Icon(
              state.season == 'Đông'
                  ? Icons.ac_unit
                  : Icons.filter_vintage_outlined,
              color: _gilt,
              size: 18,
            ),
            const SizedBox(width: 6),
            Text(
              'Mùa ${state.season}',
              style: const TextStyle(fontFamily: 'Lora', fontSize: 16),
            ),
          ],
        ),
        const SizedBox(height: 4),
        Text(
          state.seasonal
              ? 'Thiên Hòa · năm ${state.year}'
              : 'Năm ${state.year} · tuần ${(state.completedWeeks % 52) + 1}',
          style: const TextStyle(fontSize: 10, color: _gilt),
        ),
      ],
    );
    return Container(
      padding: EdgeInsets.fromLTRB(4, short ? 6 : 12, 18, short ? 6 : 12),
      decoration: const BoxDecoration(
        gradient: LinearGradient(colors: [_deepJade, _jade, _deepJade]),
        image: DecorationImage(
          image: AssetImage('assets/images/imperial/jade-texture.webp'),
          fit: BoxFit.cover,
          opacity: .25,
        ),
        border: Border(bottom: BorderSide(color: _gilt)),
      ),
      child: Column(
        children: [
          Row(
            children: [
              SizedBox(width: wide ? 220 : null, child: wide ? identity : null),
              if (!wide) Expanded(child: identity),
              if (wide) Expanded(child: _stats(short: short)),
              const SizedBox(width: 12),
              calendar,
            ],
          ),
          if (!wide) ...[const SizedBox(height: 12), _stats(short: false)],
        ],
      ),
    );
  }

  Widget _stats({required bool short}) {
    final values = <(String, int, IconData, String)>[
      ('Ngân khố', state.treasury, Icons.savings_outlined, 'treasury'),
      ('Lòng dân', state.people, Icons.groups_outlined, 'court'),
      ('Uy quyền', state.authority, Icons.workspace_premium_outlined, 'court'),
      (
        'Quân lực',
        militaryStrength(state),
        Icons.shield_outlined,
        state.population == null ? 'kingdom' : 'military',
      ),
      ('Sức khỏe', state.health, Icons.favorite_border, 'health'),
      ('Ổn định triều đình', state.stability, Icons.balance_outlined, 'court'),
    ];
    return LayoutBuilder(
      builder:
          (context, bounds) => Wrap(
            alignment: WrapAlignment.center,
            runSpacing: 6,
            children: [
              for (final stat in values)
                SizedBox(
                  width: bounds.maxWidth / (bounds.maxWidth < 440 ? 3 : 6),
                  child: Tooltip(
                    message:
                        '${stat.$1}: ${stat.$2}${stat.$4 == 'treasury'
                            ? ' ${moneyUnit(state)}'
                            : stat.$4 == 'military'
                            ? ' điểm'
                            : '/100'} · Nhấn để xem chi tiết',
                    child: InkWell(
                      key: ValueKey('citadel-stat-${stat.$1}'),
                      onTap: () => onNavigate(stat.$4),
                      borderRadius: BorderRadius.circular(6),
                      child: Padding(
                        padding: const EdgeInsets.symmetric(
                          vertical: 3,
                          horizontal: 4,
                        ),
                        child: Column(
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            Row(
                              mainAxisAlignment: MainAxisAlignment.center,
                              children: [
                                ImperialIcon(switch (stat.$1) {
                                  'Ngân khố' => 'treasury',
                                  'Lòng dân' => 'citizens',
                                  'Uy quyền' => 'authority',
                                  'Quân lực' => 'army',
                                  'Sức khỏe' => 'health',
                                  _ => 'stability',
                                }, size: short ? 24 : 30),
                                const SizedBox(width: 6),
                                Flexible(
                                  child: Text(
                                    stat.$4 == 'treasury' && state.goldStandard
                                        ? moneyNumber(stat.$2)
                                        : '${stat.$2}',
                                    maxLines: 1,
                                    style: TextStyle(
                                      fontFamily: 'Lora',
                                      fontSize: short ? 16 : 19,
                                      color:
                                          stat.$4 != 'treasury' && stat.$2 <= 20
                                              ? const Color(0xFFFFAA90)
                                              : _ivory,
                                    ),
                                  ),
                                ),
                              ],
                            ),
                            const SizedBox(height: 3),
                            Text(
                              stat.$1,
                              textAlign: TextAlign.center,
                              style: const TextStyle(
                                fontSize: 10,
                                color: _ivory,
                              ),
                            ),
                          ],
                        ),
                      ),
                    ),
                  ),
                ),
            ],
          ),
    );
  }

  Widget _scene({required bool showLocations}) => LayoutBuilder(
    builder: (context, bounds) {
      // Match hotspot positions to the exact cover transform of the artwork.
      final imageSize =
          applyBoxFit(
            BoxFit.cover,
            const Size(1672, 941),
            bounds.biggest,
          ).destination;
      final scale = math.max(bounds.maxWidth / 1672, bounds.maxHeight / 941);
      final artWidth = 1672 * scale;
      final artHeight = 941 * scale;
      final offsetX = (imageSize.width - artWidth) / 2;
      final offsetY = (imageSize.height - artHeight) / 2;
      return Stack(
        key: const ValueKey('citadel-scene'),
        fit: StackFit.expand,
        children: [
          Image.asset(
            'assets/images/citadel-landscape.webp',
            fit: BoxFit.cover,
            excludeFromSemantics: true,
            errorBuilder:
                (_, _, _) => const ColoredBox(
                  color: _jade,
                  child: Center(
                    child: Icon(Icons.castle_outlined, color: _gilt, size: 120),
                  ),
                ),
          ),
          const IgnorePointer(
            child: DecoratedBox(
              decoration: BoxDecoration(
                gradient: LinearGradient(
                  begin: Alignment.topCenter,
                  end: Alignment.bottomCenter,
                  colors: [
                    Color(0x25102922),
                    Colors.transparent,
                    Color(0x66102922),
                  ],
                  stops: [0, .45, 1],
                ),
              ),
            ),
          ),
          if (showLocations)
            for (final location in _locations)
              Positioned(
                left: (offsetX + artWidth * location.$4.dx - 83).clamp(
                  12.0,
                  bounds.maxWidth - 178,
                ),
                top: (offsetY + artHeight * location.$4.dy - 25).clamp(
                  12.0,
                  math.max(12.0, bounds.maxHeight - 62),
                ),
                width: 166,
                child: _locationButton(location),
              ),
          if (!showLocations || bounds.maxHeight >= 350)
            Positioned(
              top: 16,
              left: 16,
              child: _petitionSlip(
                compact: !showLocations || bounds.maxHeight < 350,
              ),
            ),
          if (showLocations && bounds.maxHeight >= 350)
            Positioned(
              bottom: 14,
              left: 20,
              child: Text(
                state.war > 0
                    ? 'BIÊN CƯƠNG CÓ BIẾN · HÃY CHĂM LO QUỐC SỰ'
                    : 'GIỮ NƯỚC · GIỮ LÒNG NGƯỜI · GIỮ CHÍNH MÌNH',
                style: const TextStyle(
                  fontSize: 10,
                  color: _ivory,
                  letterSpacing: 1.6,
                  shadows: [Shadow(color: _deepJade, blurRadius: 8)],
                ),
              ),
            ),
        ],
      );
    },
  );

  Widget _petitionSlip({required bool compact}) {
    final due = state.petitions.where((p) => p.deadline <= state.week).length;
    final critical =
        [
          state.people,
          state.authority,
          state.army,
          state.health,
          state.stability,
        ].where((v) => v <= 20).length;
    return Material(
      color: Colors.transparent,
      elevation: 4,
      shadowColor: _deepJade.withValues(alpha: .4),
      borderRadius: BorderRadius.circular(3),
      child: InkWell(
        key: const ValueKey('citadel-petitions'),
        onTap: () => onNavigate('court'),
        child: Container(
          width: compact ? 210 : 246,
          padding: EdgeInsets.all(compact ? 16 : 22),
          child: ImperialSkin(
            asset: 'ivory',
            edge: 20,
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Row(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const ImperialIcon('scroll', size: 34),
                  const SizedBox(width: 10),
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        const Text(
                          'Việc cần ngự phê',
                          style: TextStyle(
                            fontFamily: 'Lora',
                            fontSize: 16,
                            color: Color(0xFF3E3526),
                          ),
                        ),
                        const SizedBox(height: 4),
                        Text(
                          state.petitions.isEmpty
                              ? 'Đã xử lý hết tấu chương'
                              : '${state.petitions.length} tấu chương chờ xử lý',
                          style: const TextStyle(
                            fontSize: 11,
                            color: Color(0xFF60492F),
                          ),
                        ),
                        if (due > 0)
                          Text(
                            '$due việc tới hạn ${state.periodName} này',
                            style: const TextStyle(fontSize: 11, color: _son),
                          ),
                        if (critical > 0)
                          Text(
                            '$critical chỉ số nguy cấp · Xem ngay',
                            style: const TextStyle(fontSize: 11, color: _son),
                          ),
                        if (!compact && state.affairsThisWeek == 0)
                          const Text(
                            'Chưa xử lý việc nước',
                            style: TextStyle(fontSize: 11, color: _son),
                          ),
                      ],
                    ),
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget _locationButton(
    (String, String, IconData, Offset) location, {
    bool compact = false,
  }) => _PalaceButton(
    key: ValueKey('citadel-location-${location.$1}'),
    onPressed: () => onNavigate(location.$1),
    label: location.$2,
    icon: location.$3,
    art: location.$1,
    count: location.$1 == 'court' ? state.petitions.length : 0,
    compact: compact,
  );

  Widget _footer({required bool wide, required bool short}) {
    final navigation = <(String, String, IconData)>[
      ('citadel', 'Hoàng thành', Icons.castle_outlined),
      ('kingdom', 'Quốc sự', Icons.landscape_outlined),
      ('people', 'Triều thần', Icons.groups_outlined),
      ('harem', 'Hậu cung', Icons.local_florist_outlined),
      ('journal', 'Sử ký', Icons.menu_book_outlined),
      ('scores', 'Thành tích', Icons.emoji_events_outlined),
    ];
    final dock = Row(
      children: [
        for (final item in navigation)
          Expanded(
            child: Tooltip(
              message: item.$2,
              child: InkWell(
                key: ValueKey('nav-${item.$1}'),
                onTap: () => onNavigate(item.$1),
                borderRadius: BorderRadius.circular(8),
                child: Container(
                  padding: EdgeInsets.symmetric(
                    vertical: short ? 8 : 14,
                    horizontal: 3,
                  ),
                  decoration:
                      item.$1 == 'citadel'
                          ? BoxDecoration(
                            gradient: LinearGradient(
                              begin: Alignment.topCenter,
                              end: Alignment.bottomCenter,
                              colors: [
                                _gilt.withValues(alpha: .03),
                                _gilt.withValues(alpha: .17),
                              ],
                            ),
                            border: const Border(
                              bottom: BorderSide(color: _gilt, width: 2),
                            ),
                          )
                          : null,
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      ImperialIcon(item.$1, size: short ? 32 : 44),
                      const SizedBox(height: 6),
                      Text(
                        item.$2,
                        maxLines: 1,
                        style: TextStyle(
                          fontFamily: 'Lora',
                          fontSize: wide ? 13 : 10,
                          color: _ivory,
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ),
          ),
      ],
    );
    final advance = Column(
      mainAxisSize: MainAxisSize.min,
      children: [
        Text(
          '${state.points} hành động còn lại',
          key: const ValueKey('citadel-actions'),
          style: const TextStyle(fontSize: 11, color: _ivory),
        ),
        const SizedBox(height: 6),
        _PalaceButton(
          key: const ValueKey('citadel-end-turn'),
          onPressed: busy ? null : onEndTurn,
          label: state.seasonal ? 'Khép mùa' : 'Sang tuần',
          icon: Icons.arrow_forward,
          primary: true,
          compact: short,
        ),
        if (state.points == 0 && state.overtime == 0)
          TextButton(
            onPressed: busy ? null : onOvertime,
            child: const Text(
              'Chong đèn làm thêm',
              style: TextStyle(color: _gilt, fontSize: 11),
            ),
          ),
      ],
    );
    return Container(
      padding: EdgeInsets.symmetric(
        horizontal: wide ? 20 : 12,
        vertical: short ? 6 : 10,
      ),
      decoration: const BoxDecoration(
        gradient: LinearGradient(colors: [_deepJade, _jade, _deepJade]),
        image: DecorationImage(
          image: AssetImage('assets/images/imperial/jade-texture.webp'),
          fit: BoxFit.cover,
          opacity: .25,
        ),
        border: Border(top: BorderSide(color: _gilt)),
      ),
      child:
          wide
              ? Row(
                children: [
                  Expanded(child: dock),
                  const SizedBox(width: 28),
                  SizedBox(width: 205, child: advance),
                ],
              )
              : Column(
                children: [
                  dock,
                  const SizedBox(height: 16),
                  SizedBox(width: double.infinity, child: advance),
                ],
              ),
    );
  }
}

class _PalaceButton extends StatefulWidget {
  const _PalaceButton({
    super.key,
    required this.onPressed,
    required this.label,
    required this.icon,
    this.art,
    this.count = 0,
    this.primary = false,
    this.compact = false,
  });
  final VoidCallback? onPressed;
  final String label;
  final IconData icon;
  final String? art;
  final int count;
  final bool primary, compact;

  @override
  State<_PalaceButton> createState() => _PalaceButtonState();
}

class _PalaceButtonState extends State<_PalaceButton> {
  bool hovered = false, focused = false;

  @override
  Widget build(BuildContext context) => MouseRegion(
    onEnter: (_) => setState(() => hovered = true),
    onExit: (_) => setState(() => hovered = false),
    child: AnimatedContainer(
      duration: const Duration(milliseconds: 140),
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(widget.primary ? 10 : 5),
        border: focused ? Border.all(color: _ivory, width: 2) : null,
        boxShadow: [
          BoxShadow(
            color:
                hovered || focused
                    ? _gilt.withValues(alpha: .35)
                    : const Color(0x66000000),
            blurRadius: hovered || focused ? 16 : 6,
            offset: const Offset(0, 3),
          ),
        ],
      ),
      child: Material(
        color: Colors.transparent,
        borderRadius: BorderRadius.circular(widget.primary ? 9 : 4),
        child: InkWell(
          onTap: widget.onPressed,
          onFocusChange: (value) => setState(() => focused = value),
          borderRadius: BorderRadius.circular(widget.primary ? 9 : 4),
          child: Opacity(
            opacity: widget.onPressed == null ? .45 : 1,
            child: ImperialSkin(
              asset: widget.primary ? 'crimson' : 'jade',
              edge: 20,
              child: ConstrainedBox(
                constraints: const BoxConstraints(minHeight: 54),
                child: Padding(
                  padding: EdgeInsets.symmetric(
                    horizontal: 12,
                    vertical: widget.compact ? 8 : 12,
                  ),
                  child: Row(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      if (widget.art != null)
                        ImperialIcon(widget.art!, size: 32)
                      else
                        Icon(widget.icon, color: _gilt, size: 22),
                      const SizedBox(width: 8),
                      Flexible(
                        child: Text(
                          widget.label,
                          textAlign: TextAlign.center,
                          style: TextStyle(
                            fontFamily: 'Lora',
                            color: _ivory,
                            fontSize: widget.primary ? 19 : 15,
                          ),
                        ),
                      ),
                      if (widget.count > 0) ...[
                        const SizedBox(width: 8),
                        Container(
                          padding: const EdgeInsets.symmetric(
                            horizontal: 6,
                            vertical: 2,
                          ),
                          decoration: BoxDecoration(
                            color: _son,
                            borderRadius: BorderRadius.circular(3),
                            border: Border.all(color: _gilt),
                          ),
                          child: Text(
                            '${widget.count}',
                            style: const TextStyle(color: _ivory, fontSize: 12),
                          ),
                        ),
                      ],
                    ],
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    ),
  );
}
